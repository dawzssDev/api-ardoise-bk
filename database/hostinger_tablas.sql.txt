1a) pending_registrations  (alta diferida: User solo tras pago)
--------------------------------------------------------------------------------
-- Flujo:
--   POST /auth/register          → guarda pendiente (sin users)
--   POST /auth/register/checkout → Stripe subscription + client_secret
--   POST /auth/register/complete → crea users + negocios si pago OK
CREATE TABLE `pending_registrations` (
  `id` bigint UNSIGNED NOT NULL AUTO_INCREMENT,
  `token` char(36) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `business_name` varchar(255) NOT NULL,
  `phone` varchar(30) NOT NULL,
  `needs_invoice` tinyint(1) NOT NULL DEFAULT 0,
  `rfc` varchar(13) NULL DEFAULT NULL,
  `legal_name` varchar(255) NULL DEFAULT NULL,
  `tax_regime` varchar(10) NULL DEFAULT NULL,
  `tax_zip` varchar(10) NULL DEFAULT NULL,
  `cfdi_use` varchar(10) NULL DEFAULT NULL,
  `stripe_customer_id` varchar(255) NULL DEFAULT NULL,
  `stripe_subscription_id` varchar(255) NULL DEFAULT NULL,
  `stripe_price_id` varchar(255) NULL DEFAULT NULL,
  `status` varchar(30) NOT NULL DEFAULT 'pending',
  `expires_at` timestamp NOT NULL,
  `completed_at` timestamp NULL DEFAULT NULL,
  `user_id` bigint UNSIGNED NULL DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `pending_registrations_token_unique` (`token`),
  KEY `pending_registrations_email_status_index` (`email`, `status`),
  KEY `pending_registrations_expires_at_index` (`expires_at`),
  KEY `pending_registrations_stripe_customer_id_index` (`stripe_customer_id`),
  KEY `pending_registrations_stripe_subscription_id_index` (`stripe_subscription_id`),
  KEY `pending_registrations_user_id_foreign` (`user_id`),
  CONSTRAINT `pending_registrations_user_id_foreign`
    FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

--------------------------------------------------------------------------------
1p) tb_tipos_venta  (público general, policía, cortesía, consumo colaborador, etc.)
--------------------------------------------------------------------------------
-- tipo_descuento: ninguno | porcentaje | monto_fijo | gratis
-- valor_descuento: requerido solo si porcentaje o monto_fijo; NULL si ninguno/gratis
-- diferir_cobro: si 1, la línea NO se cobra hoy → cuenta por cobrar
-- requiere_empleado: si 1, el front debe exigir seleccionar empleado
-- require_autori: 0 = no requiere autorización, 1 = sí requiere autorización
CREATE TABLE `tb_tipos_venta` (
  `id` bigint UNSIGNED NOT NULL AUTO_INCREMENT,
  `negocio_id` bigint UNSIGNED NOT NULL,
  `name` varchar(255) NOT NULL,
  `tipo_descuento` enum('ninguno','porcentaje','monto_fijo','gratis') NOT NULL,
  `valor_descuento` decimal(10,2) NULL DEFAULT NULL,
  `diferir_cobro` tinyint(1) NOT NULL DEFAULT 0,
  `requiere_empleado` tinyint(1) NOT NULL DEFAULT 0,
  `require_autori` tinyint UNSIGNED NOT NULL DEFAULT 0,
  `status` tinyint(1) NOT NULL DEFAULT 1,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `tb_tipos_venta_negocio_id_name_unique` (`negocio_id`, `name`),
  KEY `tb_tipos_venta_negocio_id_foreign` (`negocio_id`),
  KEY `tb_tipos_venta_negocio_id_status_index` (`negocio_id`, `status`),

-- tipo_movimiento: deposito | transferencia | retiro | venta_efectivo | venta_tarjeta
-- status detalle: 0 inactivo | 1 aceptado | 2 pendiente | 3 rechazado
--   pendiente transferencia = maestra→subcuenta (descuenta maestra al crear; abona sucursal al aceptar)
--   pendiente retiro = subcuenta→maestra (sin mover saldo al crear; descuenta sucursal + abona maestra al aceptar)
--   venta_efectivo / venta_tarjeta = corte gerencial (abona solo matriz; origen = subcuenta; status aceptado)
-- cuenta_origen_id: NULL en depósito (opcional); obligatorio en transferencia/retiro

-- cuenta_destino_id: cuenta que recibe (si no se envía, usa mae_cuenta_conta_suc_id)

  `id` bigint UNSIGNED NOT NULL AUTO_INCREMENT,

  `tipo_movimiento` varchar(20) NOT NULL,
  `cuenta_origen_id` bigint UNSIGNED NULL DEFAULT NULL,
  `monto_movimiento` decimal(12,2) NOT NULL DEFAULT 0.00,
  `cuenta_destino_id` bigint UNSIGNED NULL DEFAULT NULL,
  `descripcion_movimiento` varchar(500) NOT NULL,
  `created_by` bigint UNSIGNED NOT NULL,
  `updated_by` bigint UNSIGNED NULL DEFAULT NULL,

  KEY `mae_cta_det_cuenta_id_foreign` (`mae_cuenta_conta_suc_id`),
  KEY `mae_cta_det_origen_id_foreign` (`cuenta_origen_id`),

  KEY `mae_cta_det_updated_by_foreign` (`updated_by`),
  KEY `mae_cta_det_negocio_cuenta_deleted_index` (`negocio_id`, `mae_cuenta_conta_suc_id`, `deleted`),
