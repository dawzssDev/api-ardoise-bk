================================================================================
API ARDOISE — Tablas MySQL para Hostinger
================================================================================
Orden: ejecuta los CREATE en este orden (respeta foreign keys).

RECOMENDADO: en el servidor usa `php artisan migrate --force` en lugar de
crear tablas a mano. Este archivo es solo referencia / respaldo.

Charset sugerido: utf8mb4 / utf8mb4_unicode_ci
================================================================================

--   sucursal            → sucursal_id
--   rol                 → role_id
--   empleadoResponsable → empleado_id
--   status              → status
--   fechacreacion       → created_at
--   usercreation        → created_by
--   fechaupdate         → updated_at
--   userupdate          → updated_by
CREATE TABLE `staff` (
  `id` bigint UNSIGNED NOT NULL AUTO_INCREMENT,
  `negocio_id` bigint UNSIGNED NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `sucursal_id` bigint UNSIGNED NOT NULL,
  `role_id` bigint UNSIGNED NOT NULL,
  `empleado_id` bigint UNSIGNED NOT NULL,
  `status` tinyint(1) NOT NULL DEFAULT 1,
  `created_by` bigint UNSIGNED NOT NULL,
  `updated_by` bigint UNSIGNED NULL DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `staff_negocio_id_username_unique` (`negocio_id`, `username`),
  UNIQUE KEY `staff_empleado_id_unique` (`empleado_id`),
  KEY `staff_negocio_id_foreign` (`negocio_id`),
  KEY `staff_sucursal_id_foreign` (`sucursal_id`),
  KEY `staff_role_id_foreign` (`role_id`),
  KEY `staff_empleado_id_foreign` (`empleado_id`),
  KEY `staff_created_by_foreign` (`created_by`),
  KEY `staff_updated_by_foreign` (`updated_by`),
  KEY `staff_negocio_id_status_index` (`negocio_id`, `status`),
  KEY `staff_negocio_id_sucursal_id_index` (`negocio_id`, `sucursal_id`),
  KEY `staff_negocio_id_role_id_index` (`negocio_id`, `role_id`),
  CONSTRAINT `staff_negocio_id_foreign`
    FOREIGN KEY (`negocio_id`) REFERENCES `negocios` (`id`) ON DELETE CASCADE,
  CONSTRAINT `staff_sucursal_id_foreign`
    FOREIGN KEY (`sucursal_id`) REFERENCES `sucursales` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `staff_role_id_foreign`
    FOREIGN KEY (`role_id`) REFERENCES `roles` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `staff_empleado_id_foreign`
    FOREIGN KEY (`empleado_id`) REFERENCES `empleados` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `staff_created_by_foreign`
    FOREIGN KEY (`created_by`) REFERENCES `users` (`id`) ON DELETE CASCADE,
  CONSTRAINT `staff_updated_by_foreign`
    FOREIGN KEY (`updated_by`) REFERENCES `users` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

================================================================================
NOTA SI USAS php artisan migrate
================================================================================
Si creas las tablas a mano, Laravel no sabrá que las migraciones ya corrieron.
En ese caso, o bien:

  A) NO crees tablas a mano y solo ejecuta:
     php artisan migrate --force

  B) Si ya las creaste a mano, inserta filas en `migrations` con los nombres
     de cada archivo de database/migrations, por ejemplo:

INSERT INTO `migrations` (`migration`, `batch`) VALUES
('2026_08_01_175500_add_trial_ends_at_to_subscriptions_table', 1);
